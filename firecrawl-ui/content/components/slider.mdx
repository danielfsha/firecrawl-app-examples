# Slider

A control that allows the user to select a value within a range by dragging a thumb along a track.

<Preview code={`import { Slider } from "@/components/ui/slider";

<Slider defaultValue={[50]} max={100} step={1} />`}>
  <div style={{ width: "100%", maxWidth: "320px" }}>
    <Slider defaultValue={[50]} max={100} step={1} />
  </div>
</Preview>

import { Slider as SliderPrimitive } from "@base-ui/react/slider";
import { cn } from "@/lib/utils";

function Slider({
  className,
  defaultValue,
  value,
  min = 0,
  max = 100,
  ...props
}: SliderPrimitive.Root.Props) {
  const _values = Array.isArray(value)
    ? value
    : Array.isArray(defaultValue)
      ? defaultValue
      : [min, max];

  return (
    <SliderPrimitive.Root
      className={cn("data-horizontal:w-full data-vertical:h-full", className)}
      data-slot="slider"
      defaultValue={defaultValue}
      value={value}
      min={min}
      max={max}
      thumbAlignment="edge"
      {...props}
    >
      <SliderPrimitive.Control className="relative flex w-full touch-none items-center select-none data-disabled:opacity-50">
        <SliderPrimitive.Track
          className="relative grow overflow-hidden rounded-full bg-input select-none shadow-[inset_0_1px_3px_0_rgba(0,0,0,0.02)] data-horizontal:h-2 data-horizontal:w-full"
        >
          <SliderPrimitive.Indicator
            className="bg-primary select-none data-horizontal:h-full"
          />
        </SliderPrimitive.Track>
        {Array.from({ length: _values.length }, (_, index) => (
          <SliderPrimitive.Thumb
            key={index}
            className="block h-5 w-8 shrink-0 rounded-full bg-background ring-0 transition-[color,box-shadow] select-none shadow-[0_1px_2px_0_rgba(0,0,0,0.06),0_1px_3px_0_rgba(0,0,0,0.1)]"
          />
        ))}
      </SliderPrimitive.Control>
    </SliderPrimitive.Root>
  );
}

export { Slider };`} language="tsx" showLineNumbers={true} />

### Default

<Preview code={`<Slider defaultValue={[50]} max={100} step={1} />`}>

### Range

Use an array with two values to create a range slider.

<Preview code={`<Slider defaultValue={[25, 75]} max={100} step={1} />`}>
  <div style={{ width: "100%", maxWidth: "320px" }}>
    <Slider defaultValue={[25, 75]} max={100} step={1} />
  </div>
</Preview>
